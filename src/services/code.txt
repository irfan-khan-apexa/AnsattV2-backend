/* ------------------------------------------------ */
function section(title: string) {
  console.log("\n==================================================");
  console.log(title);
  console.log("==================================================");
}
/* ------------------------------------------------ */

/* ---------------- Skill Dictionary ---------------- */

const SKILL_DICTIONARY: Record<string, string[]> = {
  "Node.js": ["nodejs", "node.js", "node js", "node"],
  "Docker": ["docker", "container", "containerization"],
  "AWS": ["aws", "amazon web services", "ec2", "s3"],
  "Redis": ["redis", "redis cache"],
  "Kubernetes": ["kubernetes", "k8s"],
};

/* ---------------- Normalize ---------------- */

function normalize(text: string) {
  return text
    .toLowerCase()
    .replace(/[^\w\s]/g, "")
    .replace(/\s+/g, " ");
}

/* ---------------- Extract Skills (UNCHANGED) ---------------- */

function extractSkills(text: string) {

  const normalized = normalize(text);

  const found: string[] = [];

  Object.entries(SKILL_DICTIONARY).forEach(([skill, variants]) => {
    variants.forEach((v) => {
      if (normalized.includes(normalize(v))) {
        found.push(skill);
      }
    });
  });

  return [...new Set(found)];
}

/* ------------------------------------------------ */
/* EXPERIENCE */
/* ------------------------------------------------ */

function extractExperience(text: string) {

  section("EXPERIENCE DETECTION");

  const match = text.match(/(\d+)\+?\s*(years|yrs)/i);

  const exp = match ? parseInt(match[1]) : 0;

  console.log("Experience detected:", exp);

  return exp;
}

/* ------------------------------------------------ */
/* PROJECT IMPACT */
/* ------------------------------------------------ */

function detectProjectImpact(text: string) {

  section("PROJECT IMPACT");

  const verbs = [
    "built","developed","implemented","designed",
    "optimized","engineered","created","improved",
    "integrated","automated","scaled","deployed",
    "configured","managed","architected"
  ];

  const clean = normalize(text);

  let score = 0;

  verbs.forEach(v => {
    if (clean.includes(v)) {
      score += 3;
    }
  });

  score = Math.min(score, 20);

  console.log("Project impact score:", score);

  return score;
}

/* ------------------------------------------------ */
/* ATS SCORE */
/* ------------------------------------------------ */

function calculateScore(
  foundSkills: string[],
  requiredSkills: string[],
  experience: number,
  minExp: number,
  maxExp: number,
  projectImpact: number
) {

  section("ATS SCORE CALCULATION");

  const skillMatch =
    foundSkills.filter(s => requiredSkills.includes(s));

  const skillScore =
    requiredSkills.length
      ? (skillMatch.length / requiredSkills.length) * 50
      : 0;

  let expScore = 0;

  if (experience >= minExp && experience <= maxExp) {
    expScore = 25;
  }
  else if (experience > maxExp) {
    expScore = 20;
  }
  else if (experience >= minExp * 0.5) {
    expScore = 10;
  }

  const extraScore =
    Math.min((foundSkills.length - skillMatch.length) * 2, 15);

  const finalScore =
    Math.min(
      Math.round(skillScore + expScore + extraScore + projectImpact),
      100
    );

  console.log("Skill score:", Math.round(skillScore));
  console.log("Experience score:", expScore);
  console.log("Extra score:", extraScore);
  console.log("Project score:", projectImpact);
  console.log("Final ATS score:", finalScore);

  return finalScore;
}

/* ------------------------------------------------ */
/* RANKING */
/* ------------------------------------------------ */

async function rankCandidates(jobId: number) {

  section("RANKING CANDIDATES");

  const candidates =
    await JobApplication.findAll({
      where: { job_id: jobId },
      order: [
        ["match_score", "DESC"],
        ["createdAt", "ASC"]
      ]
    });

  let rank = 1;

  for (const c of candidates as any[]) {

    await c.update({ rank });

    console.log(
      `Candidate ${c.id} → Rank ${rank} (Score ${c.match_score})`
    );

    rank++;

  }
}

  const normalizedText = normalize(text);

  console.log("Matched skill count:", matched.length);
  console.log("Matched skills:", matched);

  return matched;
}

/* ------------------------------------------------ */
/* ATS SCORE */
/* ------------------------------------------------ */

function calculateScore(
  foundSkills: string[],
  requiredSkills: string[],
  experience: number,
  minExp: number,
  maxExp: number,
  projectImpact: number
) {

  section("ATS SCORE CALCULATION");

  const skillMatch =
    foundSkills.filter(s => requiredSkills.includes(s));

  /* ---------------- SKILL SCORE (40%) ---------------- */

  const skillScore =
    requiredSkills.length
      ? (skillMatch.length / requiredSkills.length) * 40
      : 0;

  /* ---------------- EXPERIENCE SCORE (20%) ---------------- */

  let expScore = 0;

  if (experience >= minExp && experience <= maxExp) {
    expScore = 20;
  }
  else if (experience > maxExp) {
    expScore = 16;
  }
  else if (experience >= minExp * 0.5) {
    expScore = 8;
  }

  /* ---------------- EXTRA SCORE (15%) ---------------- */

  const extraScore =
    Math.min((foundSkills.length - skillMatch.length) * 2, 15);

  /* ---------------- PROJECT SCORE (25%) ---------------- */

  const projectScore =
    Math.min(projectImpact, 25);

  /* ---------------- FINAL SCORE ---------------- */

  const finalScore =
    Math.min(
      Math.round(
        skillScore +
        expScore +
        extraScore +
        projectScore
      ),
      100
    );

  console.log("Skill score:", Math.round(skillScore));
  console.log("Experience score:", expScore);
  console.log("Extra score:", extraScore);
  console.log("Project score:", projectScore);
  console.log("Final ATS score:", finalScore);

  return finalScore;
}
